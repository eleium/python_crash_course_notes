1. RocketGame (主类)
def __init__(self):
    pygame.init()
    self.screen = pygame.display.set_mode((1200, 700))
    self.bg_color = (230, 230, 230)
    pygame.display.set_caption("Rocket Game")
    self.clock = pygame.time.Clock()
    self.rocket = Rocket(self) # 把主类自己传给火箭

def run_game(self):
    while True:
        self._check_events()
        self.rocket.update() # 先让火箭算好位置
        self._update_screen()
        self.clock.tick(60)

def _check_events(self):
    for event in pygame.event.get():
        if event.type == pygame.QUIT: sys.exit()
        elif event.type == pygame.KEYDOWN:
            if event.key == pygame.K_UP: self.rocket.moving_up = True

2. Rocket (火箭类)
def __init__(self, r_game): # 接收主类的对象 r_game
    self.screen = r_game.screen
    self.screen_rect = r_game.screen.get_rect() # 必须先拿到矩形
    
    self.image = pygame.image.load("images/rocket.bmp")
    self.rect = self.image.get_rect()
    self.rect.center = self.screen_rect.center # 放在中央
    
    # 移动设置
    self.x = float(self.rect.x)
    self.y = float(self.rect.y)
    self.speed = 1.5 # 速度变量
    
    # 移动标志
    self.moving_up = False
    self.moving_down = False
    self.moving_left = False
    self.moving_right = False

def update(self): # 专门负责按状态更新坐标
    if self.moving_up and self.rect.top > 0: 
        self.y -= self.speed
    if self.moving_down and self.rect.bottom < self.screen_rect.bottom: 
        self.y += self.speed
    if self.moving_left and self.rect.left > 0: 
        self.x -= self.speed
    if self.moving_right and self.rect.right < self.screen_rect.right: 
        self.x += self.speed
        
    # 把浮点数更新回整数的 rect
    self.rect.x = self.x
    self.rect.y = self.y

def blitme(self):
    self.screen.blit(self.image, self.rect)
